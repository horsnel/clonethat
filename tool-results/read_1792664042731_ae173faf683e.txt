     1→'use client'
     2→
     3→import { useCallback, useEffect, useMemo, useState } from 'react'
     4→import { motion } from 'framer-motion'
     5→import { toast } from 'sonner'
     6→import { Button } from '@/components/ui/button'
     7→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
     8→import { Badge } from '@/components/ui/badge'
     9→import { AnalysisPanel } from './analysis-panel'
    10→import type { ClonePageInfo, CloneMeta } from '@/lib/cloner/types'
    11→import { formatBytes, hostnameOf, type CloneResponse } from './shared'
    12→import {
    13→  AlertTriangle, Check, Code2, Copy, Download, ExternalLink,
    14→  FileArchive, Info, Layers, Loader2, Monitor, MousePointerClick,
    15→  Package, RefreshCw, ShieldAlert, Smartphone, Sparkles, SplitSquareHorizontal, Tablet, Wand2,
    16→} from 'lucide-react'
    17→
    18→interface RebuildData {
    19→  code: string
    20→  react: string | null
    21→  previewHtml: string
    22→}
    23→
    24→const REBUILD_STEPS = [
    25→  'Reading the captured DOM…',
    26→  'Extracting design tokens…',
    27→  'Rebuilding layout with Tailwind…',
    28→  'Generating React component…',
    29→  'Polishing the code…',
    30→]
    31→
    32→interface WorkspaceProps {
    33→  result: Pick<CloneResponse, 'id' | 'meta' | 'htmlPreview' | 'htmlTruncated' | 'htmlSize'>
    34→  onNewClone: () => void
    35→}
    36→
    37→type Device = 'desktop' | 'tablet' | 'mobile'
    38→
    39→const DEVICE_CLASS: Record<Device, string> = {
    40→  desktop: 'w-full',
    41→  tablet: 'w-full max-w-[768px]',
    42→  mobile: 'w-full max-w-[390px]',
    43→}
    44→
    45→function pageLabel(p: ClonePageInfo): string {
    46→  if (p.kind === 'entry') return 'Main page'
    47→  if (p.kind === 'state') return `State · ${p.interactionLabel ?? 'interaction'}`
    48→  try {
    49→    return new URL(p.sourceUrl).pathname
    50→  } catch {
    51→    return p.slug
    52→  }
    53→}
    54→
    55→export function Workspace({ result, onNewClone }: WorkspaceProps) {
    56→  const { id, meta, htmlPreview, htmlTruncated, htmlSize } = result
    57→  const pages: ClonePageInfo[] = meta.pages ?? []
    58→  const [activeSlug, setActiveSlug] = useState(pages[0]?.slug ?? 'index')
    59→  const [device, setDevice] = useState<Device>('desktop')
    60→  const [iframeKey, setIframeKey] = useState(0)
    61→  const [copied, setCopied] = useState(false)
    62→  const [zipLoading, setZipLoading] = useState(false)
    63→  const [codeLoading, setCodeLoading] = useState(false)
    64→  const [codeText, setCodeText] = useState<string | null>(null)
    65→  const [codeTruncated, setCodeTruncated] = useState(false)
    66→  const [rebuild, setRebuild] = useState<RebuildData | null>(null)
    67→  const [rebuildLoading, setRebuildLoading] = useState(false)
    68→  const [rebuildError, setRebuildError] = useState<string | null>(null)
    69→  const [rebuildStep, setRebuildStep] = useState(0)
    70→  const [rebuildView, setRebuildView] = useState<'preview' | 'html' | 'react'>('preview')
    71→  const previewBase = `/api/preview/${id}`
    72→  const previewUrl = `${previewBase}?page=${activeSlug}`
    73→  const shotUrl = `/api/screenshot/${id}?page=${activeSlug}`
    74→  const activePage = useMemo(() => pages.find((p) => p.slug === activeSlug), [pages, activeSlug])
    75→
    76→  useEffect(() => {
    77→    setActiveSlug(pages[0]?.slug ?? 'index')
    78→    setDevice('desktop')
    79→    setIframeKey((k) => k + 1)
    80→    setCodeText(htmlPreview || null)
    81→    setCodeTruncated(htmlTruncated)
    82→    setRebuild(null)
    83→    setRebuildError(null)
    84→  }, [id])
    85→
    86→  useEffect(() => {
    87→    setIframeKey((k) => k + 1)
    88→    setCodeText(null)
    89→    setCodeTruncated(false)
    90→    setRebuild(null)
    91→    setRebuildError(null)
    92→    setRebuildView('preview')
    93→  }, [activeSlug])
    94→
    95→  // rotate progress messages while the AI is working
    96→  useEffect(() => {
    97→    if (!rebuildLoading) return
    98→    const t = setInterval(() => setRebuildStep((s) => (s + 1) % REBUILD_STEPS.length), 3200)
    99→    return () => clearInterval(t)
   100→  }, [rebuildLoading])
   101→
   102→  const ensureRebuild = useCallback(async () => {
   103→    if (rebuildLoading) return
   104→    setRebuildLoading(true)
   105→    setRebuildError(null)
   106→    try {
   107→      let res = await fetch(`/api/rebuild/${id}?page=${activeSlug}`)
   108→      if (res.status === 404) {
   109→        res = await fetch(`/api/rebuild/${id}?page=${activeSlug}`, { method: 'POST' })
   110→      }
   111→      const data = await res.json()
   112→      if (!res.ok || !data.exists) throw new Error(data.error ?? 'Rebuild unavailable.')
   113→      setRebuild({ code: data.code, react: data.react, previewHtml: data.previewHtml })
   114→    } catch (e) {
   115→      setRebuildError(e instanceof Error ? e.message : 'AI rebuild failed.')
   116→    } finally {
   117→      setRebuildLoading(false)
   118→    }
   119→  }, [id, activeSlug, rebuildLoading])
   120→
   121→  const copyRebuild = async (text: string | null, label: string) => {
   122→    if (!text) return
   123→    try {
   124→      await navigator.clipboard.writeText(text)
   125→      toast.success(`${label} copied to clipboard`)
   126→    } catch {
   127→      toast.error('Could not copy to clipboard.')
   128→    }
   129→  }
   130→
   131→  const loadCode = useCallback(async () => {
   132→    if (codeText) return
   133→    setCodeLoading(true)
   134→    try {
   135→      const res = await fetch(`/api/download/${id}?format=single&page=${activeSlug}`)
   136→      if (!res.ok) throw new Error('fetch failed')
   137→      const text = await res.text()
   138→      const truncated = text.length > 300_000
   139→      setCodeText(truncated ? text.slice(0, 300_000) : text)
   140→      setCodeTruncated(truncated)
   141→    } catch {
   142→      toast.error('Could not load the HTML source.')
   143→    } finally {
   144→      setCodeLoading(false)
   145→    }
   146→  }, [id, activeSlug, codeText])
   147→
   148→  const copyHtml = async () => {
   149→    try {
   150→      const res = await fetch(`/api/download/${id}?format=single&page=${activeSlug}`)
   151→      if (!res.ok) throw new Error()
   152→      const text = await res.text()
   153→      await navigator.clipboard.writeText(text)
   154→      setCopied(true)
   155→      toast.success('Full HTML copied to clipboard')
   156→      setTimeout(() => setCopied(false), 2000)
   157→    } catch {
   158→      toast.error('Could not copy — try the download button instead.')
   159→    }
   160→  }
   161→
   162→  const downloadZip = () => {
   163→    setZipLoading(true)
   164→    toast.info('Packaging mirror ZIP…')
   165→    window.location.href = `/api/download/${id}?format=zip`
   166→    setTimeout(() => setZipLoading(false), 2500)
   167→  }
   168→
   169→  return (
   170→    <motion.section
   171→      initial={{ opacity: 0, y: 16 }}
   172→      animate={{ opacity: 1, y: 0 }}
   173→      transition={{ duration: 0.4 }}
   174→      className="mx-auto w-full max-w-7xl px-4 py-6"
   175→      aria-label="Clone workspace"
   176→    >
   177→      {/* header */}
   178→      <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   179→        <div className="min-w-0">
   180→          <div className="flex flex-wrap items-center gap-2">
   181→            <Badge className="border border-emerald-500/30 bg-emerald-500/15 text-emerald-300 hover:bg-emerald-500/15">
   182→              <Check className="mr-1 h-3 w-3" aria-hidden /> Clone ready
   183→            </Badge>
   184→            {meta.rendered && (
   185→              <Badge variant="outline" className="gap-1 border-teal-500/40 bg-teal-500/10 text-[11px] text-teal-300">
   186→                <Monitor className="h-3 w-3" aria-hidden /> Chromium-rendered
   187→              </Badge>
   188→            )}
   189→            {pages.length > 1 && (
   190→              <Badge variant="outline" className="gap-1 border-zinc-700 text-[11px] text-zinc-300">
   191→                <Layers className="h-3 w-3" aria-hidden /> {pages.length} captures
   192→              </Badge>
   193→            )}
   194→            <span className="text-xs text-zinc-500">{hostnameOf(meta.finalUrl)}</span>
   195→          </div>
   196→          <h2 className="mt-1.5 truncate text-lg font-semibold text-zinc-100 sm:text-xl">{meta.title}</h2>
   197→        </div>
   198→        <div className="flex shrink-0 flex-wrap items-center gap-2">
   199→          <Button
   200→            size="sm"
   201→            onClick={downloadZip}
   202→            disabled={zipLoading}
   203→            className="gap-1.5 bg-emerald-500 font-semibold text-zinc-950 hover:bg-emerald-400"
   204→          >
   205→            {zipLoading ? <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden /> : <FileArchive className="h-3.5 w-3.5" aria-hidden />}
   206→            Mirror ZIP
   207→          </Button>
   208→          <Button
   209→            size="sm"
   210→            variant="outline"
   211→            onClick={() => window.open(`/api/download/${id}?format=single&page=${activeSlug}`, '_blank')}
   212→            className="gap-1.5 border-zinc-700 bg-zinc-900 text-zinc-200 hover:bg-zinc-800 hover:text-zinc-100"
   213→          >
   214→            <Download className="h-3.5 w-3.5" aria-hidden />
   215→            Single HTML
   216→          </Button>
   217→          <Button
   218→            size="sm"
   219→            variant="outline"
   220→            onClick={copyHtml}
   221→            className="gap-1.5 border-zinc-700 bg-zinc-900 text-zinc-200 hover:bg-zinc-800 hover:text-zinc-100"
   222→          >
   223→            {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" aria-hidden /> : <Copy className="h-3.5 w-3.5" aria-hidden />}
   224→            Copy HTML
   225→          </Button>
   226→          <Button
   227→            size="sm"
   228→            variant="ghost"
   229→            onClick={onNewClone}
   230→            className="gap-1.5 text-zinc-400 hover:text-zinc-100"
   231→          >
   232→            <Package className="h-3.5 w-3.5" aria-hidden />
   233→            New clone
   234→          </Button>
   235→        </div>
   236→      </div>
   237→
   238→      {/* auto-heal notice */}
   239→      {meta.autoHealed && (
   240→        <div className="mb-4 flex items-start gap-3 rounded-xl border border-teal-500/30 bg-teal-500/10 p-4">
   241→          <Info className="mt-0.5 h-4 w-4 shrink-0 text-teal-400" aria-hidden />
   242→          <div className="text-sm">
   243→            <p className="font-medium text-teal-200">Auto-healing engaged</p>
   244→            <p className="mt-0.5 text-teal-200/70">{meta.autoHealed.note}</p>
   245→          </div>
   246→        </div>
   247→      )}
   248→
   249→      {/* warnings */}
   250→      {meta.warnings.length > 0 && (
   251→        <div className="mb-4 flex items-start gap-3 rounded-xl border border-amber-500/30 bg-amber-500/10 p-4">
   252→          <ShieldAlert className="mt-0.5 h-4 w-4 shrink-0 text-amber-400" aria-hidden />
   253→          <ul className="list-inside list-disc space-y-1 text-sm text-amber-200/90">
   254→            {meta.warnings.map((w) => <li key={w}>{w}</li>)}
   255→          </ul>
   256→        </div>
   257→      )}
   258→
   259→      {meta.analysis.jsRendered && (
   260→        <div className="mb-4 flex items-start gap-3 rounded-xl border border-amber-500/30 bg-amber-500/10 p-4">
   261→          <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-400" aria-hidden />
   262→          <div className="text-sm">
   263→            <p className="font-medium text-amber-200">This site renders its content with JavaScript.</p>
   264→            <p className="mt-0.5 text-amber-200/70">
   265→              The static clone may look empty. Try the <strong>“Always render”</strong> engine for browser-captured fidelity.
   266→            </p>
   267→          </div>
   268→        </div>
   269→      )}
   270→
   271→      <div className="grid gap-4 lg:grid-cols-[340px_minmax(0,1fr)]">
   272→        <aside className="order-2 lg:order-1" aria-label="Analysis">
   273→          <AnalysisPanel meta={meta} />
   274→        </aside>
   275→
   276→        <div className="order-1 min-w-0 lg:order-2">
   277→          {/* page/state selector */}
   278→          {pages.length > 1 && (
   279→            <div className="mb-3 flex flex-wrap items-center gap-1.5" role="tablist" aria-label="Captured pages">
   280→              {pages.map((p) => (
   281→                <button
   282→                  key={p.slug}
   283→                  role="tab"
   284→                  aria-selected={activeSlug === p.slug}
   285→                  onClick={() => setActiveSlug(p.slug)}
   286→                  className={`flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-xs font-medium transition-colors ${
   287→                    activeSlug === p.slug
   288→                      ? 'border-emerald-500/50 bg-emerald-500/15 text-emerald-300'
   289→                      : 'border-zinc-800 bg-zinc-900/60 text-zinc-400 hover:border-zinc-700 hover:text-zinc-200'
   290→                  }`}
   291→                >
   292→                  {p.kind === 'state' ? (
   293→                    <MousePointerClick className="h-3 w-3" aria-hidden />
   294→                  ) : (
   295→                    <Layers className="h-3 w-3" aria-hidden />
   296→                  )}
   297→                  {pageLabel(p)}
   298→                </button>
   299→              ))}
   300→            </div>
   301→          )}
   302→
   303→          <Tabs defaultValue="preview" onValueChange={(v) => v === 'code' && loadCode()}>
   304→            <TabsList className="mb-3 grid w-full max-w-[480px] grid-cols-4 border border-zinc-800 bg-zinc-900">
   305→              <TabsTrigger value="preview" className="gap-1.5 data-[state=active]:bg-zinc-800 data-[state=active]:text-emerald-300">
   306→                <Monitor className="h-3.5 w-3.5" aria-hidden /> Preview
   307→              </TabsTrigger>
   308→              <TabsTrigger value="compare" className="gap-1.5 data-[state=active]:bg-zinc-800 data-[state=active]:text-emerald-300">
   309→                <SplitSquareHorizontal className="h-3.5 w-3.5" aria-hidden /> Compare
   310→              </TabsTrigger>
   311→              <TabsTrigger value="code" className="gap-1.5 data-[state=active]:bg-zinc-800 data-[state=active]:text-emerald-300">
   312→                <Code2 className="h-3.5 w-3.5" aria-hidden /> Code
   313→              </TabsTrigger>
   314→              <TabsTrigger value="rebuild" className="gap-1.5 data-[state=active]:bg-zinc-800 data-[state=active]:text-emerald-300">
   315→                <Sparkles className="h-3.5 w-3.5" aria-hidden /> AI Rebuild
   316→              </TabsTrigger>
   317→            </TabsList>
   318→
   319→            <TabsContent value="preview" className="mt-0">
   320→              <div className="overflow-hidden rounded-xl border border-zinc-800 bg-zinc-900/60">
   321→                <div className="flex items-center justify-between gap-2 border-b border-zinc-800 bg-zinc-900/80 px-3 py-2">
   322→                  <div className="flex items-center gap-1" role="group" aria-label="Preview viewport size">
   323→                    {(
   324→                      [
   325→                        ['desktop', Monitor],
   326→                        ['tablet', Tablet],
   327→                        ['mobile', Smartphone],
   328→                      ] as const
   329→                    ).map(([d, Icon]) => (
   330→                      <button
   331→                        key={d}
   332→                        onClick={() => setDevice(d)}
   333→                        aria-label={`${d} viewport`}
   334→                        aria-pressed={device === d}
   335→                        className={`rounded-md p-2 transition-colors ${
   336→                          device === d
   337→                            ? 'bg-emerald-500/15 text-emerald-300'
   338→                            : 'text-zinc-500 hover:bg-zinc-800 hover:text-zinc-300'
   339→                        }`}
   340→                      >
   341→                        <Icon className="h-4 w-4" aria-hidden />
   342→                      </button>
   343→                    ))}
   344→                    <span className="ml-1 hidden font-mono text-[10px] text-zinc-600 sm:inline">
   345→                      {device === 'desktop' ? '1280+' : device === 'tablet' ? '768' : '390'}px
   346→                    </span>
   347→                  </div>
   348→                  <div className="flex items-center gap-1.5">
   349→                    <span className="hidden max-w-[240px] truncate rounded-md bg-zinc-950 px-2.5 py-1 font-mono text-[10px] text-zinc-500 md:inline">
   350→                      {previewUrl}
   351→                    </span>
   352→                    <button
   353→                      onClick={() => setIframeKey((k) => k + 1)}
   354→                      aria-label="Reload preview"
   355→                      className="rounded-md p-2 text-zinc-500 transition-colors hover:bg-zinc-800 hover:text-zinc-300"
   356→                    >
   357→                      <RefreshCw className="h-4 w-4" aria-hidden />
   358→                    </button>
   359→                    <a
   360→                      href={previewUrl}
   361→                      target="_blank"
   362→                      rel="noreferrer"
   363→                      aria-label="Open preview in new tab"
   364→                      className="rounded-md p-2 text-zinc-500 transition-colors hover:bg-zinc-800 hover:text-zinc-300"
   365→                    >
   366→                      <ExternalLink className="h-4 w-4" aria-hidden />
   367→                    </a>
   368→                  </div>
   369→                </div>
   370→                <div className="flex justify-center bg-zinc-950 p-0 sm:p-4">
   371→                  <iframe
   372→                    key={`${iframeKey}-${activeSlug}`}
   373→                    src={previewUrl}
   374→                    title={`Cloned preview${activePage ? ` of ${activePage.title}` : ''}`}
   375→                    className={`h-[62vh] min-h-[440px] border-0 bg-white transition-all duration-300 sm:rounded-lg sm:shadow-2xl sm:shadow-black/60 ${DEVICE_CLASS[device]}`}
   376→                  />
   377→                </div>
   378→              </div>
   379→            </TabsContent>
   380→
   381→            <TabsContent value="compare" className="mt-0">
   382→              <div className="overflow-hidden rounded-xl border border-zinc-800 bg-zinc-900/60">
   383→                <div className="flex items-center justify-between border-b border-zinc-800 bg-zinc-900/80 px-4 py-2">
   384→                  <span className="text-xs text-zinc-400">Original (live screenshot) vs Clone (served locally)</span>
   385→                  <a
   386→                    href={shotUrl}
   387→                    target="_blank"
   388→                    rel="noreferrer"
   389→                    className="text-[11px] text-emerald-400 hover:text-emerald-300"
   390→                  >
   391→                    Open original ↗
   392→                  </a>
   393→                </div>
   394→                <div className="grid gap-0 bg-zinc-950 md:grid-cols-2">
   395→                  <div className="border-b border-zinc-800 p-3 md:border-b-0 md:border-r">
   396→                    <p className="mb-2 text-center text-[10px] uppercase tracking-wider text-zinc-600">Original</p>
   397→                    <div className="flex h-[52vh] min-h-[360px] items-start justify-center overflow-hidden rounded-lg border border-zinc-800 bg-white">
   398→                      <img
   399→                        src={shotUrl}
   400→                        alt={`Original screenshot of ${activePage?.title ?? meta.title}`}
   401→                        className="w-full object-cover object-top"
   402→                      />
   403→                    </div>
   404→                  </div>
   405→                  <div className="p-3">
   406→                    <p className="mb-2 text-center text-[10px] uppercase tracking-wider text-zinc-600">Clone</p>
   407→                    <div className="flex h-[52vh] min-h-[360px] items-start justify-center overflow-hidden rounded-lg border border-zinc-800 bg-white">
   408→                      <iframe
   409→                        key={`cmp-${iframeKey}-${activeSlug}`}
   410→                        src={previewUrl}
   411→                        title={`Clone comparison of ${activePage?.title ?? meta.title}`}
   412→                        className="h-full w-full border-0"
   413→                      />
   414→                    </div>
   415→                  </div>
   416→                </div>
   417→              </div>
   418→            </TabsContent>
   419→
   420→            <TabsContent value="code" className="mt-0">
   421→              <div className="overflow-hidden rounded-xl border border-zinc-800 bg-zinc-950">
   422→                <div className="flex items-center justify-between border-b border-zinc-800 bg-zinc-900/80 px-4 py-2">
   423→                  <span className="font-mono text-xs text-zinc-500">
   424→                    {activeSlug}.html
   425→                    {codeText && ` · ${formatBytes(new Blob([codeText]).size)}`}
   426→                    {codeTruncated && ' · showing first 300 KB'}
   427→                  </span>
   428→                  {codeLoading ? (
   429→                    <Loader2 className="h-3.5 w-3.5 animate-spin text-zinc-500" aria-hidden />
   430→                  ) : (
   431→                    <button
   432→                      onClick={copyHtml}
   433→                      aria-label="Copy HTML source"
   434→                      className="rounded-md p-1.5 text-zinc-500 transition-colors hover:bg-zinc-800 hover:text-zinc-300"
   435→                    >
   436→                      <Copy className="h-3.5 w-3.5" aria-hidden />
   437→                    </button>
   438→                  )}
   439→                </div>
   440→                <pre className="cloner-scroll max-h-[62vh] min-h-[440px] overflow-auto p-4 font-mono text-[11px] leading-relaxed text-zinc-400">
   441→                  <code>{codeLoading ? 'Loading source…' : codeText ?? 'Open the Code tab to load the source.'}</code>
   442→                </pre>
   443→              </div>
   444→            </TabsContent>
   445→
   446→            <TabsContent value="rebuild" className="mt-0">
   447→              <div className="overflow-hidden rounded-xl border border-zinc-800 bg-zinc-900/60">
   448→                {rebuildLoading && (
   449→                  <div className="flex h-[62vh] min-h-[440px] flex-col items-center justify-center gap-4 bg-zinc-950">
   450→                    <div className="relative">
   451→                      <Sparkles className="h-10 w-10 animate-pulse text-emerald-400" aria-hidden />
   452→                      <span className="absolute -inset-4 animate-ping rounded-full border border-emerald-500/20" aria-hidden />
   453→                    </div>
   454→                    <p className="text-sm font-medium text-zinc-200">{REBUILD_STEPS[rebuildStep]}</p>
   455→                    <p className="max-w-sm text-center text-xs text-zinc-500">
   456→                      The AI is rewriting the captured page as clean semantic code — this typically takes 1–3 minutes.
   457→                    </p>
   458→                  </div>
   459→                )}
   460→
   461→                {!rebuildLoading && rebuildError && (
   462→                  <div className="flex h-[62vh] min-h-[440px] flex-col items-center justify-center gap-4 bg-zinc-950 p-6">
